cmake_minimum_required(VERSION 3.21)
project(MouseMotionLabNative LANGUAGES CXX)

set(MML_ONNXRUNTIME_ROOT "" CACHE PATH "Path to the pinned ONNX Runtime SDK")

add_library(mousemotionlab_native_version STATIC src/version_stub.cpp)
target_compile_features(mousemotionlab_native_version PUBLIC cxx_std_20)
target_include_directories(mousemotionlab_native_version PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)

if (WIN32)
  add_library(mousemotionlab_pca_runtime STATIC mousegen_runtime/src/pca_runtime.cpp)
  target_compile_features(mousemotionlab_pca_runtime PUBLIC cxx_std_20)
  target_compile_definitions(mousemotionlab_pca_runtime PUBLIC MML_MOUSEGEN_STATIC WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(mousemotionlab_pca_runtime PUBLIC mousegen_runtime/include)
  target_link_libraries(mousemotionlab_pca_runtime PRIVATE bcrypt)
  add_executable(mousegen_pca_cli mousegen_runtime/cli/main.cpp)
  target_compile_features(mousegen_pca_cli PRIVATE cxx_std_20)
  target_link_libraries(mousegen_pca_cli PRIVATE mousemotionlab_pca_runtime)

  add_library(mousemotionlab_mouse_io SHARED mouse_io/src/mouse_io.cpp)
  target_compile_features(mousemotionlab_mouse_io PUBLIC cxx_std_20)
  target_compile_definitions(mousemotionlab_mouse_io PRIVATE MML_MOUSE_IO_EXPORTS UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
  target_include_directories(mousemotionlab_mouse_io PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/mouse_io/include)
  target_link_libraries(mousemotionlab_mouse_io PRIVATE user32)

  if (MML_ONNXRUNTIME_ROOT AND EXISTS "${MML_ONNXRUNTIME_ROOT}/include/onnxruntime_cxx_api.h")
    add_library(mousemotionlab_mousegen SHARED mousegen_runtime/src/mousegen_runtime.cpp)
    target_compile_features(mousemotionlab_mousegen PUBLIC cxx_std_20)
    target_compile_definitions(mousemotionlab_mousegen PRIVATE MML_MOUSEGEN_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(mousemotionlab_mousegen PUBLIC mousegen_runtime/include PRIVATE "${MML_ONNXRUNTIME_ROOT}/include")
    target_link_libraries(mousemotionlab_mousegen PRIVATE "${MML_ONNXRUNTIME_ROOT}/lib/onnxruntime.lib" bcrypt)
    add_custom_command(TARGET mousemotionlab_mousegen POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MML_ONNXRUNTIME_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:mousemotionlab_mousegen>")
    add_executable(mousegen_cli mousegen_runtime/cli/main.cpp)
    target_compile_features(mousegen_cli PRIVATE cxx_std_20)
    target_link_libraries(mousegen_cli PRIVATE mousemotionlab_mousegen)
  else()
    message(STATUS "ONNX Runtime SDK not configured; portable generator targets are disabled")
  endif()

  include(CTest)
  if (BUILD_TESTING)
    add_executable(mousemotionlab_mouse_io_tests mouse_io/tests/mouse_io_tests.cpp)
    target_compile_features(mousemotionlab_mouse_io_tests PRIVATE cxx_std_20)
    target_link_libraries(mousemotionlab_mouse_io_tests PRIVATE mousemotionlab_mouse_io)
    add_test(NAME mousemotionlab_mouse_io_tests COMMAND mousemotionlab_mouse_io_tests)
    if (TARGET mousemotionlab_mousegen)
      add_executable(mousemotionlab_mousegen_tests mousegen_runtime/tests/mousegen_runtime_tests.cpp)
      target_compile_features(mousemotionlab_mousegen_tests PRIVATE cxx_std_20)
      target_link_libraries(mousemotionlab_mousegen_tests PRIVATE mousemotionlab_mousegen)
      add_test(NAME mousemotionlab_mousegen_tests COMMAND mousemotionlab_mousegen_tests)
    endif()
    add_executable(mousemotionlab_pca_runtime_tests mousegen_runtime/tests/pca_runtime_tests.cpp)
    target_compile_features(mousemotionlab_pca_runtime_tests PRIVATE cxx_std_20)
    target_link_libraries(mousemotionlab_pca_runtime_tests PRIVATE mousemotionlab_pca_runtime)
    add_test(NAME mousemotionlab_pca_runtime_tests COMMAND mousemotionlab_pca_runtime_tests)
  endif()
endif()
